---
title: Cone
description: Two thousand six hundred opaque scales, and the form is what they hide of each other.
---

A seed cone on bare paper, and the dust it rests in.

Nothing draws the cone. It is 2,600 scales on a phyllotactic spiral. Each one is an opaque tile with its own hatch angle, spacing and lip. Each hides part of the scales behind it. The pile of them is the form. A drawn outline would be a lesser thing.

The light rakes from the left. Bare paper and crescents lie on the lit belly. Ruled hatch runs through the turn, crossed hatch covers the flank, and the tip goes dark. At arm's length it is one weight. At reading distance it is 2,600 marks, and no two are the same.

Uses: `t.times`, `t.scatter`, `t.noise`, `polygon` with `opaque` and a hatch `fill`, `circle`, `stroke`, `rect`, `t.bounds`.

```ts live
// Cone — a seed cone built one scale at a time.
//
// Every scale is its own opaque tile with its own hatching, its own shadow lip
// and its own umbo. Nothing here is a texture. Up close it is two thousand
// small decisions; from across the room it is one lit form.
import { sketch, polygon, circle, rect, stroke, fill, mm } from 'occlude';
const TAU = Math.PI * 2;
const GOLDEN = Math.PI * (3 - Math.sqrt(5));
const clamp01 = (v) => (v < 0 ? 0 : v > 1 ? 1 : v);
const smooth = (a, b, v) => {
    const u = clamp01((v - a) / (b - a));
    return u * u * (3 - 2 * u);
};
export default sketch({ aspect: [10, 13], seed: 17 }, (t) => {
    const { w, h } = t.bounds();
    const cx = w * 0.5;
    const top = h * 0.165;
    const H = h * 0.615;
    const WMAX = w * 0.235;
    // profile of the body of revolution: blunt shoulder, long taper to the tip
    const NORM = Math.pow(0.413, 0.55) * Math.pow(0.587, 0.78);
    const W = (v) => v <= 0 || v >= 1 ? 0 : (WMAX * Math.pow(v, 0.55) * Math.pow(1 - v, 0.78)) / NORM;
    const N = 2600;
    const V0 = 0.05;
    const vOf = (k) => V0 + (1 - V0) * ((k + 0.5) / N);
    const seat = (k) => {
        const v = vOf(k);
        const a = k * GOLDEN + 0.2 * t.noise(v * 2.2, 7.3) + 0.07 * t.noise(v * 6.4, 2.4);
        const r = W(v);
        const y = top + v * H;
        return { x: cx + r * Math.sin(a), y, z: Math.cos(a), X: r * Math.sin(a), Y: y, Z: r * Math.cos(a), v, a };
    };
    const seats = t.times(N, (k) => seat(k));
    const at = (k, off) => seats[k + off >= N || k + off < 0 ? k - off : k + off];
    // raking light from the left, barely in front: the form has to be modelled,
    // not lit flat
    const L = (() => {
        const l = [-0.86, -0.36, 0.31];
        const m = Math.hypot(l[0], l[1], l[2]);
        return [l[0] / m, l[1] / m, l[2] / m];
    })();
    const shadowAng = Math.atan2(-L[1], -L[0]);
    const normal = (v, a) => {
        const dv = 0.004;
        const slope = (W(Math.min(1, v + dv)) - W(Math.max(0, v - dv))) / (2 * dv);
        const m = Math.hypot(H, -slope);
        return [(H / m) * Math.sin(a), -slope / m, (H / m) * Math.cos(a)];
    };
    const rounded = (ang, p) => {
        const c = Math.abs(Math.cos(ang)), s = Math.abs(Math.sin(ang));
        return Math.pow(Math.pow(c, p) + Math.pow(s, p), -1 / p);
    };
    // the scale's own cell: its two shortest steps across the surface. Search
    // for them; do not assume the lattice, because the jitter breaks it.
    const WINDOW = 110;
    const tile = (k) => {
        const s = seats[k];
        const cands = [];
        for (let j = Math.max(0, k - WINDOW); j <= Math.min(N - 1, k + WINDOW); j++) {
            if (j === k)
                continue;
            const n = seats[j];
            const dx = n.X - s.X, dy = n.Y - s.Y, dz = n.Z - s.Z;
            const d3 = Math.hypot(dx, dy, dz);
            cands.push({ d3, v3: [dx, dy, dz], e: [(n.x - s.x) * 0.5, (n.y - s.y) * 0.5] });
        }
        cands.sort((a, b) => a.d3 - b.d3);
        const c1 = cands[0];
        if (!c1)
            return null;
        const c2 = cands.slice(1, 14).find((c) => {
            const dot = (c1.v3[0] * c.v3[0] + c1.v3[1] * c.v3[1] + c1.v3[2] * c.v3[2]) / (c1.d3 * c.d3 || 1);
            return Math.abs(dot) < 0.72;
        });
        if (!c2)
            return null;
        return { s, e1: c1.e, e2: c2.e };
    };
    const tiles = t.times(N, (k) => tile(k));
    const scales = t.times(N, (k) => k)
        .filter((k) => seats[k].z > -0.5)
        .sort((a, b) => seats[a].z - seats[b].z)
        .map((k) => {
        const cell = tiles[k];
        if (!cell)
            return null;
        const { s, e1, e2 } = cell;
        const area = Math.abs(e1[0] * e2[1] - e1[1] * e2[0]);
        if (area < 0.02)
            return null;
        const M = 26;
        const pt = (ang) => {
            const f = rounded(ang, 1.55);
            const cu = f * Math.cos(ang), su = f * Math.sin(ang);
            return [s.x + e1[0] * cu + e2[0] * su, s.y + e1[1] * cu + e2[1] * su];
        };
        const loop = t.times(M, (i) => pt((i / M) * TAU));
        const nv = normal(s.v, s.a);
        const lam = nv[0] * L[0] + nv[1] * L[1] + nv[2] * L[2];
        // a little light comes back off the ground at the far rim, so the form
        // does not simply die into the paper edge
        const rim = 0.06 * smooth(0.5, 0.98, nv[0]) * smooth(0.35, -0.4, nv[2]);
        const grain = 0.5 + 0.5 * t.noise(s.x * 0.34 + 30, s.y * 0.34);
        // an S-curve, not a power: the lit side has to reach paper and the turned
        // side has to reach black. A power can only do one end.
        const key = clamp01(0.5 + 0.56 * lam);
        const shade = clamp01(smooth(0.14, 0.86, key) * (0.9 + 0.17 * grain) + rim);
        const ang1 = (Math.atan2(e1[1], e1[0]) * 180) / Math.PI;
        const texture = shade > 0.62
            ? undefined
            : shade > 0.28
                ? fill('hatch', {
                    angle: ang1 + 58 + t.rnd(16),
                    spacing: mm(0.62 + 1.5 * ((shade - 0.28) / 0.34)),
                    align: 'shape',
                })
                : fill('crosshatch', {
                    angles: [ang1 + 56, ang1 + 126],
                    spacing: mm(0.36 + 0.95 * (shade / 0.28)),
                    align: 'shape',
                });
        // the lip: the arc of the scale turned away from the light. No two are
        // the same length or sit at quite the same angle.
        const span = (1.0 + 1.15 * (1 - shade)) * (0.88 + t.rnd(0.24));
        const skew = shadowAng + t.rnd(0.3) - 0.15;
        const steps = Math.max(5, Math.round(area * 1.8) + 5);
        const arc = (r, sp) => t.times(steps, (i) => {
            const ang = skew - sp + (2 * sp * i) / (steps - 1);
            const f = rounded(ang, 1.55) * r;
            return [
                s.x + e1[0] * f * Math.cos(ang) + e2[0] * f * Math.sin(ang),
                s.y + e1[1] * f * Math.cos(ang) + e2[1] * f * Math.sin(ang),
            ];
        });
        const lip = arc(1, span);
        return [
            polygon(loop, { opaque: true, stroke: false, pen: 'pigma-005-black', fill: texture }),
            stroke(lip, { pen: 'pigma-005-black' }),
            area > 1.2 && shade > 0.5 && circle(s, 0.16 + 0.1 * shade, { pen: 'pigma-005-black' }),
        ];
    })
        .filter(Boolean);
    // the stalk, cut short: a specimen picked up, not a plant. Drawn first, so
    // the crown of scales closes over its base.
    const stTop = top - h * 0.046;
    const stalkPts = t.times(28, (i) => {
        const u = i / 27;
        const ax = cx + 0.4 - 3.0 * u;
        const ay = top + 7 - (top + 7 - stTop) * u;
        const wid = (1.5 - 0.45 * u) * (1 + 0.14 * t.noise(u * 5.5, 11));
        return { ax, ay, wid };
    });
    const stalkLoop = [
        ...stalkPts.map((q) => [q.ax - q.wid, q.ay]),
        ...stalkPts.slice().reverse().map((q) => [q.ax + q.wid, q.ay]),
    ];
    const stalk = [
        polygon(stalkLoop, {
            opaque: true,
            pen: 'pigma-005-black',
            fill: fill('crosshatch', { angles: [70, 122], spacing: mm(0.42) }),
        }),
    ];
    // the ground: nothing but taps, pooling where the tip rests
    const base = top + H;
    const bedField = (x, y) => {
        const sx = x - cx - 9;
        const lean = sx < 0 ? sx / (w * 0.26) : sx / (w * 0.36);
        const d = Math.hypot(lean, (y - base - h * 0.018) / (h * 0.038));
        const g = Math.max(0, 1 - d);
        const contact = 0.95 * Math.max(0, 1 - Math.hypot((x - cx - 1.5) / 7, (y - base - 1.2) / 3));
        return clamp01(Math.pow(g, 1.7) * (0.4 + 0.6 * (0.5 + 0.5 * t.noise(x * 0.13, y * 0.24))) + contact);
    };
    const bed = t.scatter(bedField, { spacing: 0.46 }).points.map((p) => circle(p, 0.1 + p.density * 0.16, { pen: 'pigma-005-black' }));
    const border = rect(w * 0.06, h * 0.046, w * 0.88, h * 0.908, { pen: 'pigma-005-black' });
    return [border, bed, stalk, scales];
});
```
