title: Leaf
description: A third of a manuscript leaf, and the shadow of the part that is gone.
---

One leaf of a manuscript with about a third of it left, mounted on a board.

Someone cut the spine edge, and it holds. The fore-edge and the foot have been eaten away, and the lower outer corner is gone. Where the leaf is gone the sheet carries two marks only. One is the shadow the tear throws on the mount. The other is a dashed line of the leaf the binder once trimmed.

The text abrades before the ruling does. The ruled lines run on through passages that keep no writing at all. The initial survives because it was laid on thicker. A caption gives the shelfmark and the fraction of the leaf that remains.

Uses: `t.isolines`, `t.within`, `t.stream`, `components`, `rows`, `intersect`, `union`, `subtract`, `sdf`, `polygon` with `opaque` and hatch `fill`, `dash`, `label`, `curve`, `append`, `strokes`.

```ts live
// LEAF, RECTO — about a third of it.
//
// A written leaf on a mount. The spine was cut and is nearly straight; the
// fore-edge and the foot have been eaten, and the lower outer corner is
// simply gone. Nothing is drawn in the part that is gone except the shadow
// the tear casts on the mount and a dashed line giving the size of the whole
// leaf. The sheet is opaque, so that line is visible ONLY where the sheet is
// not: you are shown how much is missing, and only there.
//
// Everything still here is decided by one field. The damp came up at the
// foot and in at the fore-edge, and its 0.36 contour is drawn, because a
// stain has a rim and that rim is exactly where the writing stops. Inside
// it the hand is whole. Outside it the ruling is left, because a dry point
// in the skin outlives a pigment on it — so the emptiness is ruled, and can
// be counted, line by line. An older and larger hand runs under the newer
// one at a slant, and comes through only on the ground the newer one has
// given up.
//
// One passage at the foot of the second column came through the flood.
// The drawing stops where the leaf stops.
import { sketch, strokes, polygon, rect, label, curve, append, fill, mm, dash, sdf, } from 'occlude';
export default sketch({ aspect: [3, 4], seed: 11 }, (t) => {
    const FINE = 'pigma-005-black';
    const INK = 'pigma-01-black';
    const HAND = 'stabilo-88-blue';
    const clamp01 = (v) => (v < 0 ? 0 : v > 1 ? 1 : v);
    const smooth = (v) => { const s = clamp01(v); return s * s * (3 - 2 * s); };
    // ---- the leaf, as it was ---------------------------------------------
    const bx = 13, by = 10, bw = 76, bh = 113;
    const cx = bx + bw / 2, cy = by + bh / 2;
    const fore = (x) => clamp01((x - bx) / bw); // 0 at spine, 1 at fore-edge
    const foot = (y) => clamp01((y - by) / bh); // 0 at head, 1 at foot
    // ---- the leaf, as it is ----------------------------------------------
    // The spine was cut and holds. Everything else is gnawed, and the drive
    // of the gnawing rises toward the fore-edge and the foot.
    const broad = (x, y) => 0.5 + 0.5 * t.noise(x / 17 + 3, y / 22 - 7);
    const gnaw = (x, y) => {
        const at = Math.pow(fore(x), 2.1); // the fore-edge takes it
        const below = Math.pow(clamp01((foot(y) - 0.42) / 0.58), 2.0); // and so does the foot
        return (0.03 + 1.0 * at + 0.85 * below) * 20 * broad(x, y) * broad(x, y);
    };
    const worms = sdf.blend(sdf.blend(sdf.circle(61, 74, 2.3), sdf.circle(63.6, 77.4, 1.6), 2.2), sdf.circle(33, 96, 2.0), 0.1);
    const cornerGone = sdf.blend(sdf.circle(95, 112, 24), sdf.circle(72, 140, 20), 10);
    const leaf = (x, y) => sdf.box(cx, cy, bw, bh)(x, y) - gnaw(x, y);
    const raw = sdf.subtract(leaf, sdf.union(worms, cornerGone));
    // The fray is the last word on every edge, the corner loss included, so
    // no part of the boundary comes out looking cut when it was not. The
    // spine WAS cut, and there the fray is almost nothing.
    const fray = (x, y) => (0.5 + 0.5 * t.noise(x / 2.9 - 12, y / 3.2 + 5)) - 0.5
        + 0.38 * t.noise(x / 1.9 + 31, y / 2.05 - 44);
    const frayAmt = (x, y) => 0.22 + 2.1 * Math.max(Math.pow(fore(x), 1.3), Math.pow(foot(y), 1.6));
    const sheet = (x, y) => raw(x, y) - frayAmt(x, y) * fray(x, y);
    const edges = t.isolines(sheet, 0, { step: 0.3 });
    // ---- what is still legible -------------------------------------------
    // Skin near a tear is always the worst of it, and there are stains
    // besides. Ink goes first; the ruling is a groove and stays.
    const rot = (x, y) => 0.5 + 0.5 * t.noise(x / 11 - 66, y / 9 + 18);
    // The damp came in at the foot and the fore-edge and walked up the leaf.
    const damp = (x, y) => {
        const u = 0.62 * fore(x) + 1.05 * Math.pow(foot(y), 1.25);
        return clamp01(1.18 * u - 0.22 + 0.42 * t.noise(x / 21 + 5, y / 27 - 9));
    };
    // One raft of text is still there in the middle of the drowned column.
    // One passage at the foot of the second column came through the flood.
    const raft = (x, y) => clamp01(1.34 - Math.hypot((x - 65) / 15.5, (y - 92) / 12) - 0.4 * t.noise(x / 7 + 2, y / 7 - 2));
    // How much of the skin still holds ink, tear aside. Its 0.36 contour is
    // the edge of the writing, and it is drawn: a damp stain has a rim.
    const bite = (x, y) => clamp01(clamp01(1.25 - 1.55 * damp(x, y) + 1.0 * raft(x, y)) * (0.62 + 0.78 * rot(x, y)));
    const intact = (x, y) => smooth(sheet(x, y) / 8.5) * bite(x, y);
    // Ink does not fade away evenly. It is there, and then it is not, and the
    // line between the two is the tide-line of the damp.
    const inkLeft = (x, y) => smooth((intact(x, y) - 0.36) / 0.28);
    const ruleLeft = (x, y) => (0.62 + 0.38 * Math.pow(intact(x, y), 0.25)) * (1 - 0.55 * inkLeft(x, y));
    // ---- the writing ------------------------------------------------------
    const tx0 = bx + 5.5, tx1 = bx + bw - 5.0;
    const ty0 = by + 8.0, ty1 = by + bh - 8.5;
    const gut = 6.0;
    const colW = (tx1 - tx0 - gut) / 2;
    const cols = [[tx0, tx0 + colW], [tx0 + colW + gut, tx1]];
    const LEAD = 3.45;
    const lines = Math.floor((ty1 - ty0) / LEAD);
    // A versal at the head of the first column: a panel with the letter cut
    // out of it. Which letter it was is now a matter of opinion.
    const ps = LEAD * 3.9;
    const px = cols[0][0], py = ty0 - 1.4;
    const inset = ps * 0.22, th = ps * 0.098;
    const lx0 = px + inset, lx1 = px + ps - inset;
    const ly0 = py + inset, ly1 = py + ps - inset;
    const letter = sdf.union(sdf.union(sdf.segment(lx0, ly0, lx0, ly1, th), sdf.segment(lx1, ly0, lx1, ly1, th)), sdf.segment(lx0, ly0, lx1, ly1, th * 0.95));
    const flakes = sdf.blend(t.throw(rect(px, py, ps, ps), { count: 3 }).points
        .map((p) => sdf.circle(p.x, p.y, 0.25 + t.rnd(1.15))), 1.2);
    const panel = sdf.subtract(sdf.intersect((x, y) => sdf.box(px + ps / 2, py + ps / 2, ps, ps)(x, y) - 0.5 * (0.5 + 0.5 * t.noise(x / 2.2, y / 2.4)), sheet), sdf.union(letter, flakes));
    const versal = polygon(t.isolines(panel, 0, { step: 0.22 }), {
        pen: INK,
        fill: fill('crosshatch', { angles: [42, 132], spacing: mm(0.55) }),
        fillPen: FINE,
    });
    // Minuscule: a narrow comb of minims sitting on the line, with a few
    // ascenders and tails. It is a hand, not a wave — up-stroke, short flat
    // head, down-stroke, and a hairline along the baseline to the next.
    const pen = t.stream('pen');
    const scribble = (x0, x1, base, xh, s, alive = inkLeft) => {
        const out = [];
        let run = [];
        const flush = () => { if (run.length > 3)
            out.push(run); run = []; };
        const sl = xh * 0.13;
        let x = x0;
        while (x < x1) {
            const word = Math.min(x1 - x, xh * s.rnd(1.6, 6.0));
            if (word < xh * 1.0)
                break;
            const stop = x + word;
            while (x < stop) {
                const lw = xh * s.rnd(0.36, 0.48);
                if (x + lw > stop)
                    break;
                if (!s.chance(alive(x, base))) {
                    flush();
                    x += lw;
                    continue;
                }
                const j = () => s.rnd(-0.05, 0.05) * xh;
                const up = s.chance(0.13) ? xh * s.rnd(1.5, 2.0) : xh * s.rnd(0.78, 1.12);
                const tail = s.chance(0.1) ? xh * s.rnd(0.4, 0.8) : 0;
                if (run.length === 0)
                    run.push([x + j(), base + j()]);
                run.push([x + sl + j(), base - up + j()]);
                run.push([x + lw * 0.44 + sl + j(), base - up * s.rnd(0.86, 1.0) + j()]);
                run.push([x + lw * 0.44 + j(), base + tail + j()]);
                run.push([x + lw + j(), base + j()]);
                x += lw;
            }
            flush();
            x += xh * s.rnd(0.34, 0.62);
        }
        flush();
        return out;
    };
    const runs = [];
    for (let i = 0; i < lines; i++) {
        const base = ty0 + i * LEAD + LEAD * 0.72;
        if (i === 9 || i === 10 || i === 22)
            continue; // the breaks between entries
        for (let c = 0; c < 2; c++) {
            let [a, b] = cols[c];
            if (c === 0 && base < py + ps + 0.6)
                a = px + ps + 1.3; // the versal holds the head
            if (pen.chance(0.24))
                b -= pen.rnd(1.2, (b - a) * 0.55); // a name is as long as it is
            if (i === 11 || i === 23) {
                const wide = pen.rnd(0.22, 0.4) * (b - a);
                a += wide;
                b -= wide;
            }
            if (b - a < 3)
                continue;
            if (sheet((a + b) / 2, base) < -14)
                continue;
            for (const r of scribble(a, b, base, 1.55, pen))
                runs.push(r);
        }
    }
    // ---- the older hand ---------------------------------------------------
    // Scraped back and written over, at a slant, and it comes through only
    // where the newer ink has let go.
    const ang = (-12.5 * Math.PI) / 180;
    const ca = Math.cos(ang), sa = Math.sin(ang);
    const turn = (x, y) => [cx + (x - cx) * ca - (y - cy) * sa, cy + (x - cx) * sa + (y - cy) * ca];
    const ghost = t.stream('ghost');
    const under = (x, y) => clamp01(1.85 * (0.5 + 0.5 * t.noise(x / 13 - 88, y / 10.5 + 54)) - 0.80)
        * smooth(sheet(x, y) / 5) * Math.pow(1 - inkLeft(x, y), 1.3);
    const underRuns = [];
    for (let i = -5; i < 22; i++) {
        const base = by + 2 + i * 8.6;
        const seg = scribble(bx - 16, bx + bw + 16, base, 3.6, ghost, () => 0.92);
        for (const r of seg) {
            const keep = [];
            const parts = [];
            for (const p of r) {
                const q = turn(p[0], p[1]);
                if (under(q[0], q[1]) > 0.13)
                    keep.push(q);
                else {
                    if (keep.length > 2)
                        parts.push([...keep]);
                    keep.length = 0;
                }
            }
            if (keep.length > 2)
                parts.push(keep);
            for (const p of parts)
                underRuns.push(p);
        }
    }
    // ---- the ruling -------------------------------------------------------
    const ruled = [];
    const rulePen = t.stream('rule');
    for (let i = 0; i < lines; i++) {
        const base = ty0 + i * LEAD + LEAD * 0.72;
        for (const [a, b] of cols) {
            let run = [];
            for (let x = a - 1.4; x <= b + 4.0; x += 0.55) {
                const y = base + 0.14 + 0.22 * t.noise(x / 26 + i * 0.7, i * 3.1);
                if (sheet(x, y) > 0.1 && rulePen.chance(ruleLeft(x, y)))
                    run.push([x, y]);
                else {
                    if (run.length > 1)
                        ruled.push(run);
                    run = [];
                }
            }
            if (run.length > 1)
                ruled.push(run);
        }
    }
    // The two bounding rules of the written space, which a ruler drew and
    // which therefore go all the way.
    for (const x of [cols[0][0] - 1.5, cols[0][1] + gut / 2, cols[1][1] + 1.5]) {
        let run = [];
        for (let y = ty0 - 2.5; y <= ty1 + 2.5; y += 0.55) {
            const xx = x + 0.3 * t.noise(y / 30 + x, x / 9);
            if (sheet(xx, y) > 0.1 && rulePen.chance(0.45 + 0.55 * ruleLeft(xx, y)))
                run.push([xx, y]);
            else {
                if (run.length > 1)
                    ruled.push(run);
                run = [];
            }
        }
        if (run.length > 1)
            ruled.push(run);
    }
    const lift = (rs) => (rs.length ? append(...rs.map((r) => curve(r))) : null);
    const inkM = lift(runs);
    const underM = lift(underRuns);
    const ruleM = lift(ruled);
    const cut = (m) => (m ? t.within(m, edges) : null);
    // The tide-line of the damp, which is where the writing stops.
    const rim = t.within(t.isolines(bite, 0.36, { step: 0.45 }).edges.filter((e) => !e.cut).extract(), edges);
    const tide = rim.points.rows(rim.points.components().filter((r) => r.length > 46).flatMap((r) => r.indices)).edges;
    // ---- the mount --------------------------------------------------------
    // The shadow of the tear, crowded at the edge and opening outward, and
    // the reconstruction: how large the leaf was. Both are under the sheet.
    const shade = t.isolines(sheet, [-0.26, -0.7, -1.4], { step: 0.3 });
    const ghosted = rect(bx, by, bw, bh, { pen: HAND, modifiers: [dash(mm(0.8), mm(3.4))] });
    const note = [
        label('MS 41  f. 7r', 13, 126.8, 1.9, { pen: HAND }),
        label('0.34 OF LEAF', 13, 129.6, 1.5, { pen: HAND }),
    ];
    return [
        strokes(shade.edges.filter((e) => !e.cut), { pen: FINE }),
        ghosted,
        note,
        polygon(edges, { opaque: true, stroke: false }),
        strokes(tide, { pen: FINE, modifiers: [dash(mm(2.0), mm(1.3))] }),
        strokes(cut(underM), { pen: FINE }),
        strokes(cut(ruleM), { pen: FINE }),
        strokes(cut(inkM), { pen: INK }),
        versal,
    ];
});
```
