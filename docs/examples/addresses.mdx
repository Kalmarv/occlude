---
title: Addresses
description: Eleven floor plans registered on one front door, each one eating the last.
---

Eleven plans of rooms, drawn from memory and registered on the front door. They go down oldest first, and every room the artist is sure of is opaque. Each address eats the one before it, room by room. What you see is not a record. It is the damage a record takes.

Certainty lives in the geometry and not in a line weight. A wall under guess warps before it hides anything. The wall that hides and the wall that draws are one wall. A wall that is truly lost never closes.

Two things here arrived on their own. The contours of the room the artist loved best survive as a band, because four later addresses lie across them. The bare paper at the centre of the sheet is a room forgotten inside that same flat. It draws nothing, and it hides nothing.

Uses: `polygon` with `opaque`, hatch and contour `fill`, `t.noise`, `t.rnd`, `strokes`, `mm`, `t.bounds`.

```ts live
// ELEVEN HOUSES — every floor plan I have lived in, drawn from memory on one
// sheet, all of them registered on the front door.
//
// Each move puts the door down a little way from the last, so the plans drift
// without going anywhere. They are laid down oldest first, and a room I am
// sure of is OPAQUE. So the newer years eat the older ones, room by room, and
// the record is not a record.
//
// Certainty is in the geometry, never in a line weight. A wall I am guessing
// at is warped BEFORE it hides anything, so the wall that hides and the wall
// that draws are the same wall, and a wall I have really lost does not close.
// Warmth is density: a room I liked keeps its floorboards, and the one I
// loved most gets contours, gone round and round.
//
// Two things fall out of the rules that I did not put there and will not take
// out. The room at the middle of the sheet is a room I forgot, inside the flat
// I loved best — it draws nothing, hides nothing, and comes out as bare paper
// at the exact centre of everything. And the contours of the happiest room
// survive only as a band, because four later addresses were laid across them.
//
// I drew the line from door to door once, and took it off the sheet. It said
// the order out loud, and the order is not the point.
import { sketch, stroke, polygon, fill, mm, map, dash, decimate, line, circle, } from 'occlude';
const FINE = 'pigma-005-black';
const MID = 'pigma-01-black';
const HEAVY = 'pigma-05-black';
export default sketch({ aspect: [1, 1], seed: 11 }, (t) => {
    // Door anchors, in order. Eleven doors inside one small blob of paper:
    // the drift is not a line, and you never get to choose it. `side` is the
    // wall the door is in, so the plans radiate from the pile every way.
    const HOUSES = [
        { door: [42, 56], rot: -11, w: 33, h: 28, dx: 0.30, side: 0, depth: 3, conf: 0.12, warm: 0.92, grain: 0 },
        { door: [58, 62], rot: 7, w: 29, h: 25, dx: 0.62, side: 3, depth: 3, conf: 0.18, warm: 0.78, grain: 90 },
        { door: [35, 50], rot: -4, w: 40, h: 33, dx: 0.20, side: 1, depth: 3, conf: 0.48, warm: 1.00, grain: 0 },
        { door: [65, 52], rot: 13, w: 30, h: 25, dx: 0.55, side: 2, depth: 3, conf: 0.24, warm: 0.66, grain: 90 },
        { door: [56, 60], rot: -8, w: 32, h: 27, dx: 0.30, side: 2, depth: 3, conf: 0.52, warm: 0.44, grain: 0 },
        { door: [67, 42], rot: 5, w: 26, h: 32, dx: 0.60, side: 3, depth: 2, conf: 0.30, warm: 0.80, grain: 90 },
        { door: [37, 42], rot: -14, w: 31, h: 24, dx: 0.32, side: 2, depth: 3, conf: 0.46, warm: 0.62, grain: 0 },
        { door: [54, 56], rot: 3, w: 27, h: 26, dx: 0.42, side: 1, depth: 3, conf: 0.64, warm: 0.94, grain: 45 },
        { door: [67, 47], rot: -6, w: 22, h: 20, dx: 0.46, side: 0, depth: 3, conf: 0.58, warm: 0.64, grain: 0 },
        { door: [46, 54], rot: 10, w: 25, h: 22, dx: 0.26, side: 3, depth: 3, conf: 0.76, warm: 0.70, grain: 90 },
        { door: [52, 60], rot: -2, w: 28, h: 24, dx: 0.40, side: 2, depth: 3, conf: 0.96, warm: 0.26, grain: 0 },
    ];
    const WALL = 0.82; // the gap between two rooms IS the wall
    const warmest = HOUSES.reduce((a, c, i) => (c.warm > HOUSES[a].warm ? i : a), 0);
    // ---- placing a plan --------------------------------------------------
    const S = 1.18, CX = 50, CY = 50;
    const frames = HOUSES.map((H0) => {
        const H = {
            ...H0,
            w: H0.w * S,
            h: H0.h * S,
            door: [CX + (H0.door[0] - CX) * S, CY + (H0.door[1] - 52) * S],
        };
        const a = (H.rot * Math.PI) / 180;
        const c = Math.cos(a), s = Math.sin(a);
        const door = H.side === 0 ? [H.w * H.dx, H.h]
            : H.side === 1 ? [0, H.h * H.dx]
                : H.side === 2 ? [H.w * H.dx, 0]
                    : [H.w, H.h * H.dx];
        const inward = H.side === 0 ? [0, -1] : H.side === 1 ? [1, 0] : H.side === 2 ? [0, 1] : [-1, 0];
        const place = ([x, y]) => {
            const px = x - door[0], py = y - door[1];
            return [H.door[0] + px * c - py * s, H.door[1] + px * s + py * c];
        };
        return { ...H, door2: door, inward, place, cos: c, sin: s };
    });
    // ---- the rooms -------------------------------------------------------
    const carve = (st, c, depth, out) => {
        const least = 5.4;
        if (depth <= 0) {
            out.push(c);
            return;
        }
        const vertical = c.w >= c.h ? st.rnd() < 0.86 : st.rnd() < 0.14;
        const f = st.rnd(0.33, 0.67);
        if (vertical) {
            const wa = c.w * f;
            if (wa < least || c.w - wa < least) {
                out.push(c);
                return;
            }
            carve(st, { x: c.x, y: c.y, w: wa, h: c.h }, depth - 1, out);
            carve(st, { x: c.x + wa, y: c.y, w: c.w - wa, h: c.h }, depth - 1, out);
        }
        else {
            const ha = c.h * f;
            if (ha < least || c.h - ha < least) {
                out.push(c);
                return;
            }
            carve(st, { x: c.x, y: c.y, w: c.w, h: ha }, depth - 1, out);
            carve(st, { x: c.x, y: c.y + ha, w: c.w, h: c.h - ha }, depth - 1, out);
        }
    };
    // A wall you are not sure of is bent before it hides anything, so what it
    // hides and what it draws are the same wall.
    const warp = (x, y, amp, k) => [
        x + amp * t.noise(x / 8 + k * 37, y / 8 - k * 19),
        y + amp * t.noise(x / 8 - k * 13, y / 8 + k * 47),
    ];
    const loopOf = (F, c, inset, amp, k) => {
        const x0 = c.x + inset, y0 = c.y + inset;
        const x1 = c.x + c.w - inset, y1 = c.y + c.h - inset;
        if (x1 - x0 < 1.2 || y1 - y0 < 1.2)
            return null;
        const corners = [[x0, y0], [x1, y0], [x1, y1], [x0, y1]];
        const SUB = 5;
        const pts = [];
        for (let i = 0; i < 4; i++) {
            const a = corners[i], z = corners[(i + 1) % 4];
            for (let s = 0; s < SUB; s++) {
                const u = s / SUB;
                const [wx, wy] = warp(a[0] + (z[0] - a[0]) * u, a[1] + (z[1] - a[1]) * u, amp, k);
                pts.push(F.place([wx, wy]));
            }
        }
        return pts;
    };
    // Floorboards ruled by hand in the plan's own frame, cut to the room and
    // then warped with it. Ink, not a fill, so the room stays see-through.
    const ghost = (F, c, gapMm, angle, amp, k) => {
        const step = Math.max(0.55, t.len(mm(gapMm)));
        const x0 = c.x + WALL * 0.5, y0 = c.y + WALL * 0.5;
        const x1 = c.x + c.w - WALL * 0.5, y1 = c.y + c.h - WALL * 0.5;
        if (x1 - x0 < 1.2 || y1 - y0 < 1.2)
            return [];
        const a = (angle * Math.PI) / 180;
        const ux = Math.cos(a), uy = Math.sin(a); // along a board
        const vx = -uy, vy = ux; // across the boards
        const cx = (x0 + x1) / 2, cy = (y0 + y1) / 2;
        const reach = Math.hypot(x1 - x0, y1 - y0) / 2;
        const out = [];
        for (let s = -reach; s <= reach; s += step) {
            const px = cx + vx * s, py = cy + vy * s;
            // Liang–Barsky against the room rectangle.
            let t0 = -reach, t1 = reach;
            const clip = (p, q) => {
                if (p === 0)
                    return q >= 0;
                const r = q / p;
                if (p < 0) {
                    if (r > t1)
                        return false;
                    if (r > t0)
                        t0 = r;
                }
                else {
                    if (r < t0)
                        return false;
                    if (r < t1)
                        t1 = r;
                }
                return true;
            };
            if (!clip(-ux, px - x0) || !clip(ux, x1 - px))
                continue;
            if (!clip(-uy, py - y0) || !clip(uy, y1 - py))
                continue;
            if (t1 - t0 < 0.8)
                continue;
            out.push(stroke(t.times(5, (i, u) => {
                const s2 = t0 + (t1 - t0) * u;
                const [wx, wy] = warp(px + ux * s2, py + uy * s2, amp, k);
                return F.place([wx, wy]);
            }), { pen: FINE }));
        }
        return out;
    };
    // A run of a closed loop: what you can still walk before the memory stops.
    const run = (pts, from, frac) => {
        const n = Math.max(2, Math.round(pts.length * frac));
        return t.times(n, (i) => pts[(from + i) % pts.length]);
    };
    // ---- build ------------------------------------------------------------
    const ink = [];
    // I drew the line from door to door once and took it off the sheet. It
    // said the order out loud, and the order is not the point. The eleven
    // crosses are already in the order I lived in them.
    frames.forEach((F, k) => {
        const st = t.stream(`house-${k}`);
        const cells = [];
        carve(st, { x: 0, y: 0, w: F.w, h: F.h }, F.depth, cells);
        const fills = [];
        const walls = [];
        const detail = [];
        let stair = false;
        // The outer wall, one gap outside the rooms: the perimeter reads double.
        // A plan I am sure of is opaque whole — it takes the paper back from
        // everywhere I lived before it, walls, floorboards and all.
        const shell = loopOf(F, { x: 0, y: 0, w: F.w, h: F.h }, -WALL * 0.5, (1 - F.conf) * 1.4, k);
        if (shell) {
            // Only the flat I am in now takes a whole block of the sheet. Every
            // other plan hides room by room, so the wall gaps and the rooms I lost
            // stay open, and what was under them keeps coming up through the holes.
            if (k >= frames.length - 2)
                fills.push(polygon(shell, { opaque: true, stroke: false }));
            if (F.conf > 0.35) {
                walls.push(stroke([...shell, shell[0]], { pen: F.conf > 0.6 ? HEAVY : MID }));
            }
            else {
                walls.push(stroke(run(shell, Math.floor(st.rnd(20)), st.rnd(0.5, 0.8)), {
                    pen: FINE, modifiers: [dash(mm(2.6), mm(1.4))],
                }));
            }
        }
        cells.forEach((c, j) => {
            // A flat with a notch in it. Or a room that is simply gone.
            const forgot = st.rnd() < 0.38 - F.conf * 0.3;
            if (forgot)
                return;
            const n1 = t.noise(c.x / 9 + k * 5, c.y / 9 - k * 3);
            const n2 = t.noise(c.y / 7 - k * 9, c.x / 7 + k * 7);
            const conf = Math.min(1, Math.max(0, F.conf * (0.78 + 0.42 * n1)));
            const warm = Math.min(1, Math.max(0, F.warm * (0.74 + 0.5 * n2)));
            const amp = (1 - conf) * 1.5;
            const pts = loopOf(F, c, WALL * 0.5, amp, k);
            if (!pts)
                return;
            const opaque = conf > 0.34;
            const dense = warm > 0.36;
            const treasure = k === warmest && warm > 0.82;
            const gap = Math.max(0.5, map(Math.min(warm, 1), 0.36, 1, 2.4, 0.55));
            if (opaque) {
                const texture = treasure
                    ? fill('contour', { spacing: mm(0.72) })
                    : dense
                        ? fill('hatch', { angle: F.rot + F.grain, spacing: mm(gap) })
                        : undefined;
                fills.push(polygon(pts, { opaque: true, stroke: false, fill: texture, fillPen: FINE }));
            }
            else if (warm > 0.42) {
                // A room I can still see but cannot place. It keeps its floor, and it
                // hides nothing: every later wall runs straight through it. Ruled by
                // hand, because a fill in this engine would make it opaque. Wide, so
                // it stays a memory of a floor and not a piece of tone.
                fills.push(ghost(F, c, Math.max(1.55, gap * 2.4), F.grain, amp, k));
            }
            if (conf > 0.3) {
                walls.push(stroke([...pts, pts[0]], { pen: conf > 0.62 ? MID : FINE }));
            }
            else {
                walls.push(stroke(run(pts, Math.floor(st.rnd(20)), st.rnd(0.45, 0.85)), {
                    pen: FINE,
                    modifiers: [dash(mm(2.2), mm(1.5)), decimate(0.18)],
                }));
            }
            // One stair for each plan that had one. A stair is the thing you
            // remember with your legs, so it survives a move better than a room.
            if (conf > 0.5 && !stair) {
                stair = true;
                const wide = Math.min(c.w, c.h) * 0.42;
                const long = Math.max(c.w, c.h) * 0.66;
                const flat = c.w >= c.h;
                const cx = c.x + c.w * 0.5, cy = c.y + c.h * 0.5;
                const treads = 9;
                for (let s = 0; s <= treads; s++) {
                    const u = s / treads - 0.5;
                    const a = flat
                        ? F.place([cx + u * long, cy - wide / 2])
                        : F.place([cx - wide / 2, cy + u * long]);
                    const z = flat
                        ? F.place([cx + u * long, cy + wide / 2])
                        : F.place([cx + wide / 2, cy + u * long]);
                    detail.push(line(a[0], a[1], z[0], z[1], { pen: FINE }));
                }
            }
        });
        // The door swings inward, if I am sure which way it opened.
        const d = F.door2;
        const swingR = 3.6;
        if (F.conf > 0.45) {
            const inA = Math.atan2(F.inward[1], F.inward[0]);
            const arc = t.times(15, (i, u) => F.place([
                d[0] + Math.cos(inA + u * Math.PI * 0.5) * swingR,
                d[1] + Math.sin(inA + u * Math.PI * 0.5) * swingR,
            ]));
            detail.push(stroke(arc, { pen: FINE }));
            const leaf0 = F.place(d);
            const leaf1 = F.place([d[0] + F.inward[0] * swingR, d[1] + F.inward[1] * swingR]);
            detail.push(line(leaf0[0], leaf0[1], leaf1[0], leaf1[1], { pen: MID }));
        }
        ink.push(fills, walls, detail);
    });
    // Eleven doors. Whatever else went, these stayed, and they are the only
    // marks on the sheet that nothing is allowed to cover.
    ink.push(frames.map((H, k) => [
        line(H.door[0] - 1.9, H.door[1], H.door[0] + 1.9, H.door[1], { pen: HEAVY }),
        line(H.door[0], H.door[1] - 1.9, H.door[0], H.door[1] + 1.9, { pen: HEAVY }),
        k === frames.length - 1 ? circle(H.door[0], H.door[1], 0.9, { pen: HEAVY }) : null,
    ]));
    return ink;
});
```
